        function runSearchEngine() {

            const categoryVal = document.getElementById('search-category').value;

            const rawKeyword = document.getElementById('search-keyword').value.trim();

 

            const filtered = searchKnowledgeBase.filter(item => {

                const matchCategory = (categoryVal === 'all' || item.category === categoryVal);

                if (!matchCategory) return false;

 

                if (!rawKeyword) return true;

 

                const combinedText = (

                    item.title + " " +

                    item.condition + " " +

                    item.limitValue + " " +

                    item.categoryName + " " +

                    (item.note || "")

                ).toLowerCase();

 

                const normalizedQuery = rawKeyword

                    .replace(/[,，]/g, ' ')

                    .replace(/(\d+(?:\.\d+)?)\s*(吋|時|in|inch|inches)/gi, '$1吋')

                    .replace(/(\d+(?:\.\d+)?)\s*(cm|公分)/gi, '$1cm')

                    .replace(/(\d+(?:\.\d+)?)\s*%/gi, '$1%')

                    .toLowerCase();

 

                const queryTokens = normalizedQuery.split(/\s+/).filter(Boolean);

 

                const hasGteSevenToken = queryTokens.some(t => {

                    const dMatch = t.match(/^(\d+(?:\.\d+)?)(吋|時|in|inch|inches|cm|公分)?$/i);

                    if (dMatch) {

                        const val = parseFloat(dMatch[1]);

                        return val >= 7;

                    }

                    return false;

                });

 

                if (hasGteSevenToken && item.id === 'panel_inner_serviceable') {

                    return false;

                }

 

                const hasLessThanSevenToken = queryTokens.some(t => {

                    const dMatch = t.match(/^(\d+(?:\.\d+)?)(吋|時|in|inch|inches|cm|公分)?$/i);

                    if (dMatch) {

                        const val = parseFloat(dMatch[1]);

                        return val > 0 && val < 7;

                    }

                    return false;

                });

 

                if (hasLessThanSevenToken && item.id === 'panel_inner_unserviceable') {

                    return false;

                }

 

                return queryTokens.every(token => {

                    let numVal = null;

                    let pctVal = null;

 

                    const pctMatch = token.match(/^(\d+(?:\.\d+)?)%$/);

                    if (pctMatch) {

                        pctVal = parseFloat(pctMatch[1]);

                    }

 

                    const dimMatch = token.match(/^(\d+(?:\.\d+)?)(吋|時|in|inch|inches|cm|公分|sq|sqin|平方吋|%)?$/i);

                    if (dimMatch && !pctMatch) {

                        numVal = parseFloat(dimMatch[1]);

                    } else if (/^\d+(\.\d+)?$/.test(token)) {

                        numVal = parseFloat(token);

                    }

 

                    /* 【新增處理】底板與側軌：底板穿孔或撕裂 判定邏輯 */

                    if (item.id === 'base_tear_edge_serviceable') {

                        if (numVal !== null) {

                            if (numVal <= 2 && numVal > 0) return true;

                            if (numVal >= 3) return false;

                        }

                    }

 

                    if (item.id === 'base_tear_edge_unserviceable') {

                        if (numVal !== null) {

                            if (numVal >= 3) return true;

                            if (numVal <= 2 && numVal > 0) return false;

                        }

                    }

 

                    if (item.id === 'base_tear_outer_serviceable') {

                        if (numVal !== null) {

                            if (numVal >= 1 && numVal <= 11) return true;

                            if (numVal >= 12) return false;

                        }

                    }

 

                    if (item.id === 'base_tear_outer_unserviceable') {

                        if (numVal !== null) {

                            if (numVal >= 12) return true;

                            if (numVal < 12 && numVal > 0) return false;

                        }

                    }

 

                    /* 底板與側軌：ULD底板彎曲曲度判定邏輯 */

                    if (item.id === 'base_bow_loaded_serviceable') {

                        if (numVal !== null) {

                            if (numVal <= 1.24 && numVal > 0) return true;

                            if (numVal >= 1.25) return false;

                        }

                    }

 

                    if (item.id === 'base_bow_loaded_unserviceable') {

                        if (numVal !== null) {

                            if (numVal >= 1.25) return true;

                            if (numVal <= 1.24 && numVal > 0) return false;

                        }

                    }

 

                    if (item.id === 'base_bow_empty_serviceable') {

                        if (numVal !== null) {

                            if (numVal <= 1.9 && numVal > 0) return true;

                            if (numVal >= 2.0) return false;

                        }

                    }

 

                    if (item.id === 'base_bow_empty_unserviceable') {

                        if (numVal !== null) {

                            if (numVal >= 2.0) return true;

                            if (numVal <= 1.9 && numVal > 0) return false;

                        }

                    }

 

                    /* 門體與五金配件：門纜磨損與鋼絲外露判定邏輯 */

                    if (item.id === 'door_cable_serviceable') {

                        if (pctVal !== null) {

                            return pctVal >= 1 && pctVal <= 24;

                        }

                        if (numVal !== null) {

                            if (numVal >= 1 && numVal <= 11) return true;

                            if (numVal >= 12 || numVal >= 25) return false;

                        }

                    }

 

                    if (item.id === 'door_cable_unserviceable') {

                        if (pctVal !== null) {

                            return pctVal >= 25;

                        }

                        if (numVal !== null) {

                            if (numVal >= 12) return true;

                            if (numVal < 12 && numVal > 0) return false;

                        }

                    }

 

                    if (item.id === 'door_canvas_serviceable' && numVal !== null && ((numVal >= 1 && numVal <= 7) || (numVal >= 1 && numVal <= 95))) {

                        return true;

                    }

 

                    if (item.id === 'door_canvas_unserviceable' && numVal !== null && (numVal >= 8 || numVal >= 96)) {

                        return true;

                    }

 

                    if (item.id === 'panel_contour_serviceable' && numVal !== null && numVal >= 2) {

                        return false;

                    }

 

                    if (item.id === 'panel_contour_serviceable' && numVal !== null && numVal < 2 && numVal > 0) {

                        return true;

                    }

 

                    if (item.id === 'panel_contour' && numVal !== null && numVal >= 2) {

                        return true;

                    }

 

                    if (hasGteSevenToken && (item.id === 'panel_edge' || item.id === 'panel_inner_unserviceable')) {

                        return true;

                    }

 

                    if (dimMatch && !pctMatch) {

                        const unit = dimMatch[2] ? dimMatch[2].toLowerCase() : '';

 

                        if (['吋', '時', 'in', 'inch', 'inches'].includes(unit)) {

                            const inchPattern = new RegExp(`(?<!\\d)${numVal}\\s*(吋|時|in|inch)`, 'i');

                            const cmVal = (numVal * 2.54).toFixed(2).replace(/\.00$/, '');

                            const cmPattern = new RegExp(`(?<!\\d)${cmVal}\\s*cm`, 'i');

                           

                            if (inchPattern.test(combinedText) || cmPattern.test(combinedText)) return true;

                        } else if (['cm', '公分'].includes(unit)) {

                            const cmPattern = new RegExp(`(?<!\\d)${numVal}\\s*(cm|公分)`, 'i');

                            if (cmPattern.test(combinedText)) return true;

                        }

                    }

 

                    if (/^\d+(\.\d+)?$/.test(token)) {

                        const numPattern = new RegExp(`(?<![\\d.])${token}(?![\\d.])`, 'i');

                        if (numPattern.test(combinedText)) return true;

                    }

 

                    if (item.id === 'panel_inner_serviceable' && numVal !== null && numVal > 0 && numVal < 7) {

                        return true;

                    }

 

                    if (item.id === 'panel_edge' && numVal !== null && [1, 4, 5].includes(numVal)) {

                        return true;

                    }

 

                    if (/^[a-z]{1,3}$/i.test(token)) {

                        const regex = new RegExp(`\\b${token}\\b`, 'i');

                        return regex.test(combinedText);

                    }

 

                    return combinedText.includes(token);

                });

            });

 

            renderSearchResults(filtered);

        }
